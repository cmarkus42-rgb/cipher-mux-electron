```typescript
const handleOrchestratorToggle = useCallback(async () => {
  const api = (window as any).cipherMux
  try {
    const status = await api.orchestrator.status()
    if (status.running && status.sessionId) {
      // Running — stop it
      await api.orchestrator.stop()
      removeSession(status.sessionId)
      setOrchestratorSessionId(null)

  }
```

```typescript
const applyMerges = useCallback((cols: number, rows: number, merges: Record<string, true>) => {
  setGrid((prev) => {
    const newSlots = prev.slots.map((s) => ({ ...s, rowSpan: 1 }))

    for (let col = 0; col < cols; col++) {
      for (let row = 0; row < rows - 1; row++) {
        const key = `${col}:${row}`
        if (!merges[key]) continue

        // This cell merges downward — calculate total span
        const topIdx = row * cols + col
        if (topIdx >= newSlots.length) continue

        let span = 1
        let r = row
        while (r < rows - 1 && merges[`${col}:${r}`]) {
          span++
          r++
        }
        newSlots[topIdx] = { ...newSlots[topIdx], rowSpan: span }

        // Hide cells that are consumed by the merge
        for (let hiddenRow = row + 1; hiddenRow < row + span; hiddenRow++) {
          const hiddenIdx = hiddenRow * cols + col
          if (hiddenIdx < newSlots.length) {
            newSlots[hiddenIdx] = { ...newSlots[hiddenIdx], rowSpan: 0 }
          }
        }
      }
    }

    const next = { ...prev, slots: newSlots }
    persist(next)
    return next
  })
}, [persist])
```

```tsx
  const getSessionName = (sessionId: string | null) => {
    if (!sessionId) return '(empty)'
    return sessions.find((s) => s.id === sessionId)?.name ?? '?'
  }

  return (
    <div class="grid-placement-overlay" onClick={onCancel}>
      <div class="grid-placement-popup" onClick={(e) => e.stopPropagation()}>
        <div class="grid-placement-popup__title">Place session — click a slot</div>
        <div
          class="grid-placement-grid"
          style={{ gridTemplateColumns: `repeat(${cols}, 1fr)` }}
        >
          {gridSlots.map((slot, idx) => {
            if (slot.rowSpan === 0) return null // hidden by merge
            return (
              <div
                key={idx}
                class={`grid-placement-cell${slot.sessionId ? ' grid-placement-cell--occupied' : ''}`}
                style={{ gridRow: slot.rowSpan > 1 ? `span ${slot.rowSpan}` : undefined }}
                onClick={() => onSelect(idx)}
                title={slot.sessionId ? `Replace: ${getSessionName(slot.sessionId)}` : 'Empty slot'}
              >
                <span class="grid-placement-cell__label">
                  {getSessionName(slot.sessionId)}
                </span>
              </div>
            )
          })}
        </div>
      </div>
    </div>
  )
}
```

```typescript
private sidebarWindow: BrowserWindow | null = null

openSidebarWindow(): void {
  if (this.sidebarWindow && !this.sidebarWindow.isDestroyed()) {
    this.sidebarWindow.focus()
    return
  }

  this.sidebarWindow = new BrowserWindow({
    width: 320,
    height: 600,
    minWidth: 250,
    minHeight: 300,
    frame: true,
    titleBarStyle: 'hiddenInset',
    webPreferences: {
      preload: path.join(__dirname, 'preload.js'),
      contextIsolation: true,
      nodeIntegration: false,
      sandbox: false,
    },
  })

  const url = this.isDev
    ? `http://localhost:5173/index.html?view=sidebar`
    : `file://${path.join(__dirname, '../renderer/index.html')}?view=sidebar`

  this.sidebarWindow.loadURL(url)
  this.sidebarWindow.on('closed', () => {
    this.sidebarWindow = null
    this.sendToMainWindow('sidebar:reattached', {})
  })
}

closeSidebarWindow(): void {
  if (this.sidebarWindow && !this.sidebarWindow.isDestroyed()) {
    this.sidebarWindow.close()
  }
}

sendToSidebarWindow(channel: string, data: unknown): void {
  if (this.sidebarWindow && !this.sidebarWindow.isDestroyed()) {
    this.sidebarWindow.webContents.send(channel, data)
  }
}
```
